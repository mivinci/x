file(GLOB_RECURSE XDNS_HEADERS "*.h")
file(GLOB_RECURSE XDNS_SOURCES "*.c")
list(FILTER XDNS_HEADERS EXCLUDE REGEX "_test")
list(FILTER XDNS_SOURCES EXCLUDE REGEX "_test")
add_library(xdns ${X_LIB_TYPE} ${XDNS_SOURCES} ${XDNS_HEADERS})
x_source_group(xdns "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(xdns PROPERTIES FOLDER "libx")

target_link_libraries(xdns PUBLIC xbase)

if(WIN32)
  target_link_libraries(xdns PRIVATE iphlpapi ws2_32)
  target_compile_definitions(xdns PRIVATE _WIN32_WINNT=0x0601 _CRT_SECURE_NO_WARNINGS)
endif()

# ── Tests ──────────────────────────────────────────────────────────────
option(XDNS_BUILD_TESTS "Build xdns tests" ${X_BUILD_TESTS})
if(XDNS_BUILD_TESTS)
  file(GLOB_RECURSE XDNS_TEST_SOURCES "*_test.cpp")
  if(XDNS_TEST_SOURCES)
    add_executable(xdns_test ${XDNS_TEST_SOURCES})
    target_link_libraries(xdns_test PRIVATE xdns GTest::gtest_main)
    x_source_group(xdns_test "${CMAKE_CURRENT_SOURCE_DIR}")
    set_target_properties(xdns_test PROPERTIES FOLDER "libx")
    if(WIN32)
      target_link_libraries(xdns_test PRIVATE ws2_32 iphlpapi)
      target_compile_options(xdns_test PRIVATE /std:c++20)
      target_compile_definitions(xdns_test PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    add_test(NAME xdns_test COMMAND xdns_test)
  endif()
endif()
