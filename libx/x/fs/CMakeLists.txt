file(GLOB_RECURSE XFS_HEADERS "*.h")
file(GLOB_RECURSE XFS_SOURCES "*.c")

add_library(xfs ${X_LIB_TYPE} ${XFS_SOURCES} ${XFS_HEADERS})
set_target_properties(xfs PROPERTIES FOLDER "libx")
target_link_libraries(xfs PUBLIC xbase)

option(XFS_BUILD_TESTS "Build xfs tests" ${X_BUILD_TESTS})
if(XFS_BUILD_TESTS)
  file(GLOB_RECURSE XFS_TESTS "*_test.cpp")
  list(FILTER XFS_TESTS EXCLUDE REGEX "_bench")
  if(XFS_TESTS)
    add_executable(xfs_test ${XFS_TESTS})
    target_link_libraries(xfs_test PRIVATE xfs GTest::gtest_main)
    set_target_properties(xfs_test PROPERTIES FOLDER "libx")
    if(WIN32)
      target_link_libraries(xfs_test PRIVATE ws2_32 iphlpapi)
      target_compile_options(xfs_test PRIVATE /std:c++20)
      target_compile_definitions(xfs_test PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    add_test(NAME xfs_test COMMAND xfs_test)
  endif()
endif()
